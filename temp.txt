    print(f'DATASET DATE RANGE: {min_date} to {max_date}')
    
    # 4. Count resolved complaints for Health Dept (All Time vs Last 30 Days relative to max_date or CURRENT_DATE)
    count_all = conn.execute(text(\"\"\"
        SELECT COUNT(*) 
        FROM complaint c
        WHERE c.department_id IN :dept_ids
          AND c.status_id IN :status_ids;
    \"\"\"), {'dept_ids': dept_ids, 'status_ids': status_ids}).scalar()
    print('TOTAL RESOLVED HEALTH COMPLAINTS (ALL TIME):', count_all)

    # Relative to MAX(created_at) in DB
    count_30_rel = conn.execute(text(\"\"\"
        SELECT COUNT(*) 
        FROM complaint c
        WHERE c.department_id IN :dept_ids
          AND c.status_id IN :status_ids
          AND c.created_at >= (SELECT MAX(created_at) FROM complaint) - INTERVAL '30 days';
    \"\"\"), {'dept_ids': dept_ids, 'status_ids': status_ids}).scalar()
    print('RESOLVED HEALTH COMPLAINTS IN LAST 30 DAYS (RELATIVE TO MAX DB DATE):', count_30_rel)

    # Relative to CURRENT_DATE
    count_30_curr = conn.execute(text(\"\"\"
        SELECT COUNT(*) 
        FROM complaint c
        WHERE c.department_id IN :dept_ids
          AND c.status_id IN :status_ids
          AND c.created_at >= CURRENT_DATE - INTERVAL '30 days';
    \"\"\"), {'dept_ids': dept_ids, 'status_ids': status_ids}).scalar()
    print('RESOLVED HEALTH COMPLAINTS IN LAST 30 DAYS (RELATIVE TO TODAY):', count_30_curr)
    
    # Breakdown by Status
    breakdown = conn.execute(text(\"\"\"

        FROM complaint c
        JOIN status_master sm ON c.status_id = sm.id
        WHERE c.department_id IN :dept_ids
        GROUP BY sm.status_name, sm.status_name_mar;
    \"\"\"), {'dept_ids': dept_ids}).fetchall()
    print('\nHEALTH COMPLAINTS STATUS BREAKDOWN (ALL TIME):')
    for b in breakdown:

        SELECT

            COUNT(*) AS total_count
        FROM complaint
        WHERE department_id IN (15, 100)
        GROUP BY TO_CHAR(created_at, 'YYYY-MM')
        ORDER BY month DESC
        LIMIT 12;
    \"\"\")).fetchall()
    print('\nRECENT MONTHLY BREAKDOWN FOR HEALTH DEPT:')
    for r in recent_counts:
